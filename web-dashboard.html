📱 EzraApp Push Notification Dashboard

How to Send Notifications:

  1. Get FCM Token: Open EzraApp and check console logs for "FCM Token:"
  2. Paste Token: Copy the token and paste it in the "FCM Token" field below
  3. Compose Message: Fill in the title and message
  4. Send: Click "Send Notification" to test

🔧 Server Integration Examples

Node.js with firebase-admin:

const admin = require('firebase-admin');
const serviceAccount = require('./serviceAccountKey.json');

admin.initializeApp({
  credential: admin.credential.cert(serviceAccount)
});

async function sendNotification(token, title, body, data = {}) {
  const message = {
    token,
    notification: { title, body },
    data
  };
  
  try {
    const response = await admin.messaging().send(message);
    console.log('Successfully sent message:', response);
    return response;
  } catch (error) {
    console.error('Error sending message:', error);
    throw error;
  }
}

// Usage
sendNotification('FCM_TOKEN_HERE', 'EzraApp', 'New devotional available!');
